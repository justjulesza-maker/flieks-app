rules_version = '2';

service firebase.storage {
  match /b/{bucket}/o {

    // ============ 4flieks platform ============
    //
    // Every file the portal uploads is stamped with its uploader
    // (metadata.owner) and given a name that has never been used, so nothing
    // is ever overwritten.
    //
    // Playback, posters, trailers, scripts, headshots and link previews all
    // use the download links stored in the database. Those carry their own
    // token and are NOT affected by these rules, so the only read the app
    // needs is the uploader fetching the link for a file they just uploaded.
    // Everything else is refused. That is what stops a signed-in account
    // fetching a film's master file without paying (which the old
    // "allow read: if request.auth != null" allowed).
    //
    // Files uploaded before this change have no owner stamp, so nobody can
    // fetch or overwrite them through the app. Their stored links still work.
    // Manage them in the Firebase console, which these rules don't apply to.

    function signedIn()     { return request.auth != null; }
    // "create" must mean a file that doesn't exist yet. Without this, a write
    // over an existing, unstamped file was accepted as a create — that is how
    // an old poster got replaced in testing.
    function isNew()        { return resource == null; }
    function ownsIt()       { return signedIn() && resource.metadata.owner == request.auth.uid; }
    function stampedByMe()  { return signedIn() && request.resource.metadata.owner == request.auth.uid; }
    function underMB(n)     { return request.resource.size < n * 1024 * 1024; }
    function isImage()      { return request.resource.contentType.matches('image/.*'); }
    function isVideo()      { return request.resource.contentType.matches('video/.*'); }

    // Films, trailers, posters, link-preview images
    match /flieks_films/{filmId}/{file} {
      allow get:    if ownsIt();
      allow list:   if false;
      allow create: if isNew() && stampedByMe()
                    && ((isVideo() && request.resource.size < 2 * 1024 * 1024 * 1024)
                     || (isImage() && underMB(10)));
      allow update: if ownsIt() && stampedByMe();
      allow delete: if ownsIt();
    }

    // Podcast episodes (video or audio) and channel covers, uploaded in the
    // Podcast Studio. Playback goes through the flieks-podcasts function, which
    // only hands the link to signed-in viewers (and buyers, for paid episodes).
    function isAudio()      { return request.resource.contentType.matches('audio/.*'); }
    match /flieks_podcasts/{channelId}/{file} {
      allow get:    if ownsIt();
      allow list:   if false;
      allow create: if isNew() && stampedByMe()
                    && ((isVideo() && request.resource.size < 2 * 1024 * 1024 * 1024)
                     || (isAudio() && underMB(500))
                     || (isImage() && underMB(10)));
      allow update: if ownsIt() && stampedByMe();
      allow delete: if ownsIt();
    }

    // Screenplays (PDF). Readers get them through the flieks-script
    // function, which hands out the stored link to signed-in accounts
    // only when the filmmaker has made the script public.
    match /flieks_scripts/{filmId}/{file} {
      allow get:    if ownsIt();
      allow list:   if false;
      allow create: if isNew() && stampedByMe() && underMB(30)
                    && request.resource.contentType == 'application/pdf';
      allow update: if ownsIt() && stampedByMe();
      allow delete: if ownsIt();
    }

    // Cast and crew headshots
    match /flieks_cast/{filmId}/{file} {
      allow get:    if ownsIt();
      allow list:   if false;
      allow create: if isNew() && stampedByMe() && isImage() && underMB(10);
      allow update: if ownsIt() && stampedByMe();
      allow delete: if ownsIt();
    }

    // Old poster folder. Nothing in the app writes here any more; anything
    // still in it keeps working through its stored link.
    match /flieks_posters/{allPaths=**} {
      allow read, write: if false;
    }

    // ============ Destroying Desmond ============
    //
    // Storage rules have no precedence: if ANY matching rule allows a request,
    // it is allowed. The old catch-all "match /dd/{allPaths=**} { allow read:
    // if true }" therefore also matched dd/film/..., and anyone — signed in or
    // not — could list the feature and fetch its download link. The catch-all
    // below is split by folder so it can never match dd/film.

    function ddAdmin() {
      return request.auth != null && request.auth.uid == 'LGANwGT28XPHyESixxX7yZiV1UQ2';
    }

    // The feature itself. Only the DD admin account can read or write it
    // through the app; dd-stream hands paying accounts the stored tokened link.
    match /dd/film/{allPaths=**} {
      allow read: if ddAdmin();
      allow write: if ddAdmin() && request.resource.size < 5 * 1024 * 1024 * 1024;
    }

    match /dd/cast/{allPaths=**} {
      allow read: if true;
      allow write: if ddAdmin() && request.resource.size < 10 * 1024 * 1024;
    }

    // Any other DD folder: public read as before, but never the film folder.
    match /dd/{folder}/{allPaths=**} {
      allow read: if folder != 'film';
      allow write: if folder != 'film' && ddAdmin()
                   && request.resource.size < 2 * 1024 * 1024 * 1024;
    }

    // Files sitting directly in dd/ (not in a folder): public read as before.
    match /dd/{file} {
      allow read: if true;
      allow write: if ddAdmin() && request.resource.size < 2 * 1024 * 1024 * 1024;
    }
  }
}
